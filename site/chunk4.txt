eateElement('div');
      el.className='weekday';
      el.textContent=w;
      grid.appendChild(el);
    });

    for(let i=0;i<firstDow;i++){
      const blank=document.createElement('div');
      blank.className='day empty';
      blank.textContent='0';
      grid.appendChild(blank);
    }

    for(let day=1;day<=daysInMonth;day++){
      const el=document.createElement('div');
      el.className='day';
      el.textContent=String(day);

      const key=`${year}-${month}-${day}`;
      const st=statusMap.get(key);
      if(st?.exists)el.classList.add('event-day');
      if(st?.planned)el.classList.add('planned-day');
      if(year===now.getFullYear() && month===now.getMonth()+1 && day===now.getDate()){
        el.classList.add('today');
      }
      if(st?.exists){
        el.title=st.planned?'参加予定イベントあり':'イベントあり';
        el.addEventListener('click',()=>scrollToEvent(year,month,day));
      }
      grid.appendChild(el);
    }

    card.appendChild(grid);
    calendarMonths.appendChild(card);
  }
}

function saveLocal(msg='自動保存済み'){
  try{
    tbody.querySelectorAll('.plan-check,.ignore-check').forEach(cb=>cb.checked?cb.setAttribute('checked','checked'):cb.removeAttribute('checked'));

    const flags={};
    const plannedByName={};
    tbody.querySelectorAll('.event-row').forEach(row=>{
      const key=eventKey(row);
      const name=row.querySelector('.event-name')?.innerText?.trim();
      const planned=!!row.querySelector('.plan-check')?.checked;
      const ignored=!!row.querySelector('.ignore-check')?.checked;
      if(key)flags[key]={planned,ignored};
      if(name)plannedByName[name]=planned;
      applyIgnoredStyle(row);
    });

    localStorage.setItem(FLAGS_KEY,JSON.stringify(flags));
    // 旧版からの移行互換用として残す
    localStorage.setItem(PLANNED_KEY,JSON.stringify(plannedByName));
    // このHTML版の手動編集だけは同じ版の再読込時に復元
    localStorage.setItem(STORAGE_KEY,tbody.innerHTML);

    status.textContent=msg;status.className='status save-ok';
  }catch(e){
    status.textContent='自動保存できませんでした';status.className='status save-warn';
  }
}
function restoreLocal(){
  try{
    // 同じHTML版で行った手動編集のみ復元。次回のHTML更新ではSTORAGE_KEYが変わるため、
    // 古い表全体が最新版を上書きすることはない。
    const s=localStorage.getItem(STORAGE_KEY);
    if(s)tbody.innerHTML=s;

    ensureEventYears();

    const flags=JSON.parse(localStorage.getItem(FLAGS_KEY)||'{}');
    const oldPlanned=JSON.parse(localStorage.getItem(PLANNED_KEY)||'{}');

    tbody.querySelectorAll('.event-row').forEach(row=>{
      const key=eventKey(row);
      const name=row.querySelector('.event-name')?.innerText?.trim();
      const plan=row.querySelector('.plan-check');
      const ignore=row.querySelector('.ignore-check');

      if(key && flags[key]){
        if(plan)plan.checked=!!flags[key].planned;
        if(ignore)ignore.checked=!!flags[key].ignored;
      }else if(name && Object.prototype.hasOwnProperty.call(oldPlanned,name)){
        // 旧版で付けた「参加予定」チェックを初回だけ引き継ぐ
        if(plan)plan.checked=!!oldPlanned[name];
      }

      // 矛盾防止：×を優先
      if(ignore?.checked && plan)plan.checked=false;
      applyIgnoredStyle(row);
    });
  }catch(e){}
}
